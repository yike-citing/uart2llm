cmake_minimum_required(VERSION 3.24)
project(uart2llm_gui VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The first release supports Windows only.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(UART2LLM_FETCH_WX "Build pinned wxWidgets source when no installed package is available" ON)
find_package(wxWidgets QUIET COMPONENTS core base)
if(wxWidgets_FOUND)
  include(${wxWidgets_USE_FILE})
  set(UART2LLM_WX_LIBRARIES ${wxWidgets_LIBRARIES})
elseif(UART2LLM_FETCH_WX)
  include(FetchContent)
  set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(wxBUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(wxBUILD_SAMPLES OFF CACHE STRING "" FORCE)
  set(wxBUILD_DEMOS OFF CACHE BOOL "" FORCE)
  set(wxUSE_WEBVIEW OFF CACHE BOOL "" FORCE)
  set(wxUSE_STC OFF CACHE BOOL "" FORCE)
  set(wxUSE_MEDIACTRL OFF CACHE BOOL "" FORCE)
  set(wxUSE_RIBBON OFF CACHE BOOL "" FORCE)
  set(wxUSE_AUI OFF CACHE BOOL "" FORCE)
  set(wxUSE_RICHTEXT OFF CACHE BOOL "" FORCE)
  set(wxUSE_PROPGRID OFF CACHE BOOL "" FORCE)
  set(wxUSE_HTML OFF CACHE BOOL "" FORCE)
  set(wxUSE_HELP OFF CACHE BOOL "" FORCE)
  set(wxUSE_WXHTML_HELP OFF CACHE BOOL "" FORCE)
  set(wxUSE_MS_HTML_HELP OFF CACHE BOOL "" FORCE)
  set(wxUSE_XRC OFF CACHE BOOL "" FORCE)
  set(wxUSE_XML OFF CACHE BOOL "" FORCE)
  set(wxUSE_REGEX OFF CACHE BOOL "" FORCE)
  set(wxUSE_LIBTIFF OFF CACHE BOOL "" FORCE)
  set(wxUSE_LIBJPEG OFF CACHE BOOL "" FORCE)
  set(wxUSE_LIBPNG builtin CACHE STRING "" FORCE)
  set(wxUSE_ZLIB builtin CACHE STRING "" FORCE)
  set(wxUSE_EXPAT OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(wxWidgets URL https://github.com/wxWidgets/wxWidgets/releases/download/v3.2.8/wxWidgets-3.2.8.zip URL_HASH SHA256=0f781c17f4a50f42b1533d6e149d6e87bb47dc51c3992144918a9c231d5615a2 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(wxWidgets)
  set(UART2LLM_WX_LIBRARIES wx::core wx::base)
else()
  message(FATAL_ERROR "Install wxWidgets or enable UART2LLM_FETCH_WX")
endif()
add_executable(uart2llm-gui WIN32 src/main.cpp src/http.cpp)
target_compile_definitions(uart2llm-gui PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(uart2llm-gui PRIVATE ${UART2LLM_WX_LIBRARIES} winhttp advapi32)
if(MINGW)
  target_link_options(uart2llm-gui PRIVATE -static)
endif()
install(TARGETS uart2llm-gui RUNTIME DESTINATION .)
include(CTest)
if(BUILD_TESTING)
  add_executable(uart2llm-native-tests tests/http_contract.cpp src/http.cpp)
  target_include_directories(uart2llm-native-tests PRIVATE src)
  target_compile_definitions(uart2llm-native-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(uart2llm-native-tests PRIVATE winhttp)
  if(MINGW)
    target_link_options(uart2llm-native-tests PRIVATE -static)
  endif()
  add_test(NAME native-http-contract COMMAND uart2llm-native-tests)
endif()
